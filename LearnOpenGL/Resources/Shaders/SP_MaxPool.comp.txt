
#version 430

layout(local_size_x = 16, local_size_y = 16, local_size_z = 1) in;

layout(r32f, binding = 0) readonly uniform image2DArray uInputImage;
layout(r32f, binding = 1) writeonly uniform image2DArray uOutputImage;


void main()
{
    ivec2 pixel = ivec2(gl_GlobalInvocationID.xy);

    int channel = int(gl_GlobalInvocationID.z);

    ivec3 inputSize = imageSize(uInputImage);
    ivec3 outputSize = imageSize(uOutputImage);

    if (pixel.x >= outputSize.x ||
        pixel.y >= outputSize.y ||
        channel >= outputSize.z)
        return;

    ivec2 inputPosition = pixel * 2;

    float maxValue = -3.402823e38;

    for (int y = 0; y < 2; y++)
    {
        for (int x = 0; x < 2; x++)
        {
            ivec2 samplePosition = inputPosition + ivec2(x, y);

            if (samplePosition.x < inputSize.x && samplePosition.y < inputSize.y)
            {
                float value = imageLoad(
                    uInputImage,
                    ivec3(samplePosition, channel)
                ).r;

                maxValue = max(maxValue, value);
            }
        }
    }

    imageStore(
        uOutputImage,
        ivec3(pixel, channel),
        vec4(maxValue, 0.0, 0.0, 0.0)
    );
}
