#version 430

layout(local_size_x = 16, local_size_y = 16, local_size_z = 1) in;

layout(r32f, binding = 0) readonly uniform image2D uInputScores;
layout(r32f, binding = 1) writeonly uniform image2D uOutputMask;

uniform int uNmsRadius;


void main()
{
    ivec2 pixel = ivec2(gl_GlobalInvocationID.xy);

    ivec2 imageSizeValue = imageSize(uInputScores);

    if (pixel.x >= imageSizeValue.x || pixel.y >= imageSizeValue.y)
        return;


    //-----------Get Center Score---------//

    float centerScore = imageLoad(uInputScores, pixel).r;

    float maxScore = -3.402823e38;


    //-----------Find Local Maximum---------//

    for (int dy = -uNmsRadius; dy <= uNmsRadius; dy++)
    {
        for (int dx = -uNmsRadius; dx <= uNmsRadius; dx++)
        {
            ivec2 neighbor = pixel + ivec2(dx, dy);

            if (neighbor.x < 0 || neighbor.x >= imageSizeValue.x ||
                neighbor.y < 0 || neighbor.y >= imageSizeValue.y)
                continue;

            float neighborScore = imageLoad(
                uInputScores,
                neighbor
            ).r;

            maxScore = max(maxScore, neighborScore);
        }
    }


    //-----------Generate Maximum Mask---------//

    float maxMask = 0.0;

    if (centerScore == maxScore)
    {
        maxMask = 1.0;
    }

    imageStore(
        uOutputMask,
        pixel,
        vec4(maxMask, 0.0, 0.0, 0.0)
    );
}
