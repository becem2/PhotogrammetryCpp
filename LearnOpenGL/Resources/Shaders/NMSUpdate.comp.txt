
#version 430

layout(local_size_x = 16, local_size_y = 16, local_size_z = 1) in;

layout(r32f, binding = 0) readonly uniform image2D uInputScores;
layout(r32f, binding = 1) readonly uniform image2D uCurrentMask;
layout(r32f, binding = 2) readonly uniform image2D uSuppressionMask;
layout(r32f, binding = 3) writeonly uniform image2D uNextMask;

uniform int uNmsRadius;


void main()
{
    ivec2 pixel = ivec2(gl_GlobalInvocationID.xy);

    ivec2 imageSizeValue = imageSize(uInputScores);

    if (pixel.x >= imageSizeValue.x || pixel.y >= imageSizeValue.y)
        return;


    //-----------Read Current Masks---------//

    float currentMask = imageLoad(
        uCurrentMask,
        pixel
    ).r;

    float suppression = imageLoad(
        uSuppressionMask,
        pixel
    ).r;


    //-----------Calculate Suppressed Center Score---------//

    float centerScore = imageLoad(
        uInputScores,
        pixel
    ).r;

    if (suppression > 0.5)
    {
        centerScore = 0.0;
    }


    //-----------Find Maximum Suppressed Score---------//

    float maxScore = -3.402823e38;

    for (int dy = -uNmsRadius; dy <= uNmsRadius; dy++)
    {
        for (int dx = -uNmsRadius; dx <= uNmsRadius; dx++)
        {
            ivec2 neighbor = pixel + ivec2(dx, dy);

            if (neighbor.x < 0 || neighbor.x >= imageSizeValue.x ||
                neighbor.y < 0 || neighbor.y >= imageSizeValue.y)
                continue;


            float neighborScore = imageLoad(
                uInputScores,
                neighbor
            ).r;

            float neighborSuppression = imageLoad(
                uSuppressionMask,
                neighbor
            ).r;

            if (neighborSuppression > 0.5)
            {
                neighborScore = 0.0;
            }

            maxScore = max(maxScore, neighborScore);
        }
    }


    //-----------Update Maximum Mask---------//

    bool newMaximum = (centerScore == maxScore);

    float nextMask = 0.0;

    if (currentMask > 0.5 ||
        (newMaximum && suppression <= 0.5))
    {
        nextMask = 1.0;
    }


    imageStore(
        uNextMask,
        pixel,
        vec4(nextMask, 0.0, 0.0, 0.0)
    );
}
