#version 430

layout(local_size_x = 16, local_size_y = 16, local_size_z = 1) in;

layout(rgba8, binding = 0) readonly uniform image2D uInputImage;
layout(rgba8, binding = 1) writeonly uniform image2D uGrayImage;

void main()
{
    ivec2 pixel = ivec2(gl_GlobalInvocationID.xy);

    ivec2 imageSizeValue = imageSize(uInputImage);

    if (pixel.x >= imageSizeValue.x || pixel.y >= imageSizeValue.y)
        return;

    vec3 rgb = imageLoad(uInputImage, pixel).rgb;

    float gray = dot(rgb, vec3(0.299, 0.587, 0.114));

    imageStore(uGrayImage, pixel, vec4(gray, gray, gray, 1.0));
}
