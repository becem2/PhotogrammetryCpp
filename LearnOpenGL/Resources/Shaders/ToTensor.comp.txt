#version 430

layout(local_size_x = 16, local_size_y = 16, local_size_z = 1) in;

layout(rgba8, binding = 0) readonly uniform image2D uGrayImage;
layout(r32f, binding = 1) writeonly uniform image2D uTensorImage;

void main()
{
    ivec2 pixel = ivec2(gl_GlobalInvocationID.xy);

    ivec2 imageSizeValue = imageSize(uGrayImage);

    if (pixel.x >= imageSizeValue.x || pixel.y >= imageSizeValue.y)
        return;

    float value = imageLoad(uGrayImage, pixel).r;

    imageStore(uTensorImage, pixel, vec4(value, 0.0, 0.0, 0.0));
}
