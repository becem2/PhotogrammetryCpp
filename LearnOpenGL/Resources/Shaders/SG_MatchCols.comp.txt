#version 430

// Column-wise arg-max of the final log assignment matrix (dustbins excluded):
//
//   indices1[j] = argmax_i P[i][j],   maxScores1[j] = max_i P[i][j]

layout(local_size_x = 64) in;

layout(std430, binding = 0) readonly buffer CouplingBuffer { float coupling[]; };
layout(std430, binding = 1) readonly buffer UBuffer { float u[]; };
layout(std430, binding = 2) readonly buffer VBuffer { float v[]; };
layout(std430, binding = 3) writeonly buffer IndexBuffer { int indices[]; };
layout(std430, binding = 4) writeonly buffer ScoreBuffer { float maxScores[]; };

uniform int uM;
uniform int uN;

void main()
{
    const int j = int(gl_GlobalInvocationID.x);   // 0 .. N-1

    if (j >= uN)
    {
        return;
    }

    const int stride = uN + 1;

    float best = -1e30;
    int bestIndex = 0;

    for (int i = 0; i < uM; i++)
    {
        const float x = coupling[i * stride + j] + u[i];

        if (x > best)
        {
            best = x;
            bestIndex = i;
        }
    }

    indices[j] = bestIndex;
    maxScores[j] = best + v[j] + log(float(uM + uN));
}
