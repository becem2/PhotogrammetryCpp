#version 430 core

layout(local_size_x = 16, local_size_y = 16, local_size_z = 1) in;


//-----------Input / Output Textures---------//

layout(r32f, binding = 0) readonly uniform image2DArray uConv;
layout(r32f, binding = 1) writeonly uniform image2DArray uRelu;


//-----------Main---------//

void main()
{
    //-----------Get Invocation Coordinates---------//

    ivec3 pixel = ivec3(gl_GlobalInvocationID);

    ivec3 size = imageSize(uConv);

    if (any(greaterThanEqual(pixel, size)))
    {
        return;
    }


    //-----------ReLU---------//

    float value = imageLoad(uConv, pixel).r;

    float result = max(value, 0.0);


    //-----------Store Output---------//

    imageStore(uRelu, pixel, vec4(result, 0.0, 0.0, 0.0));
}
