
#version 430

layout(local_size_x = 16, local_size_y = 16, local_size_z = 1) in;

layout(r32f, binding = 0) readonly uniform image2DArray uDetectorLogits;
layout(r32f, binding = 1) writeonly uniform image2D uHeatmap;


void main()
{
    ivec2 pixel = ivec2(gl_GlobalInvocationID.xy);

    ivec3 inputSize = imageSize(uDetectorLogits);
    ivec2 outputSize = imageSize(uHeatmap);

    if (pixel.x >= inputSize.x || pixel.y >= inputSize.y)
        return;

    if (inputSize.z != 65)
        return;

    if (outputSize.x != inputSize.x * 8 ||
        outputSize.y != inputSize.y * 8)
        return;


    //-----------Find Maximum Logit---------//

    float maxLogit = -3.402823e38;

    for (int channel = 0; channel < 65; channel++)
    {
        float logit = imageLoad(
            uDetectorLogits,
            ivec3(pixel, channel)
        ).r;

        maxLogit = max(maxLogit, logit);
    }


    //-----------Calculate Softmax Denominator---------//

    float sumExp = 0.0;

    for (int channel = 0; channel < 65; channel++)
    {
        float logit = imageLoad(
            uDetectorLogits,
            ivec3(pixel, channel)
        ).r;

        sumExp += exp(logit - maxLogit);
    }


    //-----------Remove Dustbin + Reconstruct Heatmap---------//

    for (int channel = 0; channel < 64; channel++)
    {
        float logit = imageLoad(
            uDetectorLogits,
            ivec3(pixel, channel)
        ).r;

        float probability = exp(logit - maxLogit) / sumExp;

        int dx = channel % 8;
        int dy = channel / 8;

        ivec2 outputPosition = pixel * 8 + ivec2(dx, dy);

        imageStore(
            uHeatmap,
            outputPosition,
            vec4(probability, 0.0, 0.0, 0.0)
        );
    }
}
